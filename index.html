async function sendAutoWhatsApp() { // هذه البيانات تحصل عليها عند فتح حساب تجريبي مجاني const instance_id = "instance12345"; // رقم المعرف الخاص بك const token = "abc123token"; // مفتاح الأمان const phone = "9639xxxxxxxx"; // رقم المستلم const message = "*نشرة طقس أمجد مريم* 🌦️\nالحرارة في دمشق: 4°C"; const url = `https://api.ultramsg.com/${instance_id}/messages/chat`; // إرسال الطلب بنظام POST (تلقائي بالكامل) const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded' }, body: `token=${token}&to=${phone}&body=${encodeURIComponent(message)}` }); if(response.ok) { alert("✅ وصلت الرسالة للواتساب تلقائياً!"); } }
